# Define variables
$enterpriseAppObjectId = "f0a81d36-bdbb-4826-8b56-a1a31598fabd"
$appRoleId = "e18f0405-fdec-4ae8-a8a0-d8edb98b061f"
$userUPNs = @("dhunter@valenzhealth.com")

# Connect to Microsoft Graph 
Connect-MgGraph -Scopes "Application.ReadWrite.All", "User.Read.All", "Directory.ReadWrite.All"

# Loop through each user and assign them to the app
foreach ($userUPN in $userUPNs) {
    try {
        $user = Get-MgUser -UserId $userUPN

        if ($user) {
            New-MgServicePrincipalAppRoleAssignment -ServicePrincipalId $enterpriseAppObjectId `
                -PrincipalId $user.Id `
                -ResourceId $enterpriseAppObjectId `
                -AppRoleId $appRoleId

            Write-Host "Added $userUPN to the app successfully." -ForegroundColor Green
        } else {
            Write-Host "User $userUPN not found." -ForegroundColor Yellow
        }
    } catch {
        Write-Host ("Error adding " + $userUPN + ":") -ForegroundColor Red
        Write-Host $_.Exception.Message -ForegroundColor Red
    }
}
